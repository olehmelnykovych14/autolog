  const metrics=[{label:'Витрати за місяць',value:'18 400 грн',icon:'💰',trend:'+12%',up:false},{label:'Загальний пробіг',value:fmt(totalMileage)+' км',icon:'🛣️',trend:'+5%',up:true},{label:'Витрати на км',value:costPerKm+' грн',icon:'📊',trend:'-3%',up:true},{label:'Наступне ТО',value:'за 2 800 км',icon:'🔧',trend:'Скоро',up:false}]

  return (
    <div className="flex flex-col gap-6">
      <div className="grid grid-cols-1 sm:grid-cols-2 xl:grid-cols-4 gap-4">
        {metrics.map((m,i)=>(

function Garage({ cars, records, onAdd }) {
  const [showAdd,setShowAdd]=useState(false)
  const [showQR,setShowQR]=useState(false)
  const [showShare,setShowShare]=useState(false)
  const [showMaint,setShowMaint]=useState(false)
  const [selCar,setSelCar]=useState(null)
  const [carList,setCarList]=useState(cars)
  useEffect(()=>setCarList(cars),[cars])
  return (
    <div className="flex flex-col gap-6">
      <div className="flex items-center justify-between flex-wrap gap-3">
        <h2 className="text-2xl font-bold text-[#2B3674] dark:text-white">Мій гараж</h2>
        <div className="flex gap-3">
          <Btn variant="secondary" onClick={()=>{setSelCar(cars[0]);setShowQR(true)}}><QrCode size={16}/>Мій QR</Btn>
          <Btn variant="secondary" onClick={()=>{setSelCar(cars[0]);setShowShare(true)}}><Share2 size={16}/>Поділитися</Btn>
          <Btn onClick={()=>setShowAdd(true)}><Plus size={16}/>Додати авто</Btn>
        </div>
      </div>
      <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-5">
        {carList.map(car=>{
          const cr=records.filter(r=>r.carId===car.id)
          const spent=cr.reduce((s,r)=>s+r.cost,0)
          return(
            <div key={car.id} className="rounded-3xl overflow-hidden shadow-[0_18px_40px_rgba(112,144,176,0.12)] dark:shadow-none relative" style={{background:`linear-gradient(135deg,${car.color}CC,${car.color}44)`}}>
              <div className="absolute inset-0 backdrop-blur-[1px]"/>
              <div className="relative p-5 text-white">
                <div className="flex items-start justify-between mb-6">
                  <div>
                    <p className="text-3xl font-black">{car.make}</p>
                    <p className="text-lg font-semibold opacity-80">{car.model} · {car.year}</p>
                  </div>
                  <div className={cx('w-3 h-3 rounded-full mt-1',car.status==='ok'?'bg-green-400':'bg-yellow-400')}/>
                </div>
                <div className="bg-white/20 backdrop-blur rounded-2xl px-4 py-2 inline-flex items-center gap-2 mb-4">
                  <span className="text-xs">🇺🇦</span>
                  <span className="font-bold tracking-widest">{car.plate}</span>
                </div>
                <div className="grid grid-cols-2 gap-3 mb-5">
                  <div className="bg-white/15 rounded-2xl p-3"><p className="text-xs opacity-70">Пробіг</p><p className="font-bold">{fmt(car.mileage)} км</p></div>
                  <div className="bg-white/15 rounded-2xl p-3"><p className="text-xs opacity-70">Витрачено</p><p className="font-bold">{fmt(spent)} грн</p></div>
                </div>
                <button onClick={()=>{setSelCar(car);setShowMaint(true)}} className="w-full py-2.5 rounded-2xl bg-white/25 hover:bg-white/35 font-semibold text-sm transition-all backdrop-blur">🤖 AI План обслуговування</button>
              </div>
            </div>
          )
        })}
      </div>
      <AddCarModal open={showAdd} onClose={()=>setShowAdd(false)} onAdd={c=>{ onAdd(c) }}/>
      <QRModal open={showQR} onClose={()=>setShowQR(false)} car={selCar}/>
      <ShareModal open={showShare} onClose={()=>setShowShare(false)} car={selCar}/>
      <MaintenancePlanModal open={showMaint} onClose={()=>setShowMaint(false)} car={selCar}/>
    </div>
  )
}
