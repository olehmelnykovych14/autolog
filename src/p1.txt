const cx = (...a) => a.filter(Boolean).join(' ')
const fmt = (n) => n.toLocaleString('uk-UA')

function Modal({ open, onClose, title, children, wide }) {
  useEffect(() => { if (open) document.body.style.overflow = 'hidden'; else document.body.style.overflow = ''; return () => { document.body.style.overflow = '' } }, [open])
  if (!open) return null
  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
      <div className="absolute inset-0 bg-black/60 backdrop-blur-sm" onClick={onClose} />
      <div className={cx('relative z-10 bg-white dark:bg-[#111C44] rounded-3xl shadow-2xl w-full max-h-[90vh] overflow-y-auto scrollbar-thin', wide ? 'max-w-3xl' : 'max-w-md')}>
        <div className="flex items-center justify-between p-6 border-b border-gray-100 dark:border-white/5">
          <h2 className="text-lg font-bold text-[#2B3674] dark:text-white">{title}</h2>
          <button onClick={onClose} className="p-2 hover:bg-gray-100 dark:hover:bg-white/10 rounded-xl transition-all text-[#A3AED0]"><X size={18} /></button>
        </div>
        <div className="p-6">{children}</div>
      </div>
    </div>
  )
}

function Btn({ children, onClick, variant = 'primary', sm, className, disabled }) {
  const base = 'font-semibold rounded-xl transition-all duration-200 flex items-center gap-2 disabled:opacity-50'
  const size = sm ? 'px-3 py-1.5 text-sm' : 'px-5 py-2.5 text-sm'
  const v = {
    primary: 'bg-[#4318FF] hover:bg-[#3311DD] text-white shadow-lg shadow-[#4318FF]/30',
    secondary: 'bg-gray-100 dark:bg-white/10 hover:bg-gray-200 dark:hover:bg-white/20 text-[#2B3674] dark:text-white',
    danger: 'bg-red-500 hover:bg-red-600 text-white',
    ghost: 'hover:bg-gray-100 dark:hover:bg-white/10 text-[#A3AED0] hover:text-[#2B3674] dark:hover:text-white',
  }
  return <button onClick={onClick} disabled={disabled} className={cx(base, size, v[variant], className)}>{children}</button>
}

function Input({ label, value, onChange, type = 'text', placeholder }) {
  return (
    <div className="flex flex-col gap-1.5">
      {label && <label className="text-xs font-semibold text-[#A3AED0] uppercase tracking-wide">{label}</label>}
      <input type={type} value={value} onChange={e => onChange(e.target.value)} placeholder={placeholder}
        className="px-4 py-2.5 rounded-xl border border-gray-200 dark:border-white/10 bg-gray-50 dark:bg-white/5 text-[#2B3674] dark:text-white placeholder-[#A3AED0] focus:outline-none focus:ring-2 focus:ring-[#4318FF]/50 transition-all text-sm" />
    </div>
  )
}

function Select({ label, value, onChange, options }) {
  return (
    <div className="flex flex-col gap-1.5">
      {label && <label className="text-xs font-semibold text-[#A3AED0] uppercase tracking-wide">{label}</label>}
      <select value={value} onChange={e => onChange(e.target.value)}
        className="px-4 py-2.5 rounded-xl border border-gray-200 dark:border-white/10 bg-gray-50 dark:bg-white/5 text-[#2B3674] dark:text-white focus:outline-none focus:ring-2 focus:ring-[#4318FF]/50 transition-all text-sm">
        {options.map(o => <option key={o} value={o}>{o}</option>)}
      </select>
    </div>
  )
}

function Card({ children, className }) {
  return <div className={cx('bg-white dark:bg-[#111C44] rounded-2xl shadow-[0_18px_40px_rgba(112,144,176,0.12)] dark:shadow-none p-5', className)}>{children}</div>
}
